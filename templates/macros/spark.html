{# Sparkline for one Zukunftsmonitor indicator (data/monitor.toml), drawn in Tera, no JS. Same geometry everywhere (monitor tiles, pillar heroes): series as a brand-gradient line with a soft area, uncertain points hollow, the target as a dashed Swiss-red level with its year, an optional projection range as a capsule. Lower-is-better indicators are drawn inverted (up = better) so "rising" always reads as progress. #} {% macro spark(ind, lang, w=320, h=96, target_label="") %} {%- set vals = ind.series | map(attribute="value") -%} {%- set years = ind.series | map(attribute="year") -%} {%- set vmin = vals | sort | first -%}{%- set vmax = vals | sort | last -%} {%- if ind.target and not ind.lower_is_better and ind.target.value > vmax -%}{%- set vmax = ind.target.value -%}{%- endif -%} {%- if ind.target and ind.lower_is_better and ind.target.value < vmin -%}{%- set vmin = ind.target.value -%}{%- endif -%} {%- if ind.target and not ind.lower_is_better and ind.target.value < vmin -%}{%- set vmin = ind.target.value -%}{%- endif -%} {%- if ind.projection and ind.projection.max > vmax -%}{%- set vmax = ind.projection.max -%}{%- endif -%} {%- if not ind.lower_is_better and vmin > 0 and (vmin < vmax * 0.6 or vmin == vmax) -%}{%- set vmin = 0 -%}{%- endif -%} {%- set ymin = years | first -%}{%- set ymax = years | last -%} {%- if ind.target and ind.target.year > ymax -%}{%- set ymax = ind.target.year -%}{%- endif -%} {%- if ind.projection and ind.projection.year > ymax -%}{%- set ymax = ind.projection.year -%}{%- endif -%} {%- if ymax == ymin -%}{%- set ymax = ymin + 1 -%}{%- endif -%} {%- set vspan = vmax - vmin -%}{%- if vspan == 0 -%}{%- set vspan = 1 -%}{%- endif -%} {%- set yspan = ymax - ymin -%} {%- set pl = 6 -%}{%- set pr = 6 -%}{%- set pt = 14 -%}{%- set pb = 18 -%} {%- set iw = w - pl - pr -%}{%- set ih = h - pt - pb -%} {%- set gid = "sg-" ~ ind.id ~ "-" ~ w -%} {%- set_global pts = "" -%}{%- set_global x0 = 0 -%}{%- set_global xl = 0 -%}{%- set_global yl = 0 -%} {%- for p in ind.series -%} {%- set x = (p.year - ymin) / yspan * iw + pl -%} {%- set f = (p.value - vmin) / vspan -%} {%- if ind.lower_is_better -%}{%- set y = pt + f * ih -%}{%- else -%}{%- set y = pt + ih - f * ih -%}{%- endif -%} {%- set xr = x | round(precision=1) -%}{%- set yr = y | round(precision=1) -%} {%- set_global pts = pts ~ xr ~ "," ~ yr ~ " " -%}{%- set_global yl = yr -%} {%- if loop.first -%}{%- set_global x0 = x -%}{%- endif -%}{%- set_global xl = x -%} {%- endfor -%} {%- endmacro %}
